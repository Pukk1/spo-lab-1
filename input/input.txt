function sum(arg1 as int, arg2 as int) as int
    arg1 + arg2;
end function

function sub(arg1 as int, arg2 as int) as int
    arg1 - arg2;
end function

function mul(arg1 as int, arg2 as int) as int
    arg1 * arg2;
end function

function div(arg1 as int, arg2 as int) as int
    arg1 / arg2;
end function

function read_num() as int
    dim in as int
    dim res as int
    dim i as int
    res = 0;
    i = 1;
    while true
        in = stdin();
        if in == 13 then
            stdin();
            break
        end if
        res = res * 10 + (in - 48);
        i = i + 1;
    wend
    res;
end function

function number_length(num as int) as int
    dim rest as int
    dim count as int
    count = 1;
    rest = num/10;
    while rest != 0
        rest = rest/10;
        count = count + 1;
    wend
    count;
end function

function print_num(num as int) as int
    dim rest as int
    dim count as int
    dim shift as int
    rest = num;
    count = number_length(num) - 1;
    shift = 1;
    while count != 0
        shift = shift * 10;
        count = count - 1;
    wend
    while shift != 0
        stdout(rest / shift + 48);
        rest = rest % shift;
        shift = shift / 10;
    wend
    stdout(10);
    stdout(13);
    0;
end function

function strlen(s as string) as int
    dim len as int
    len = 0;
    while s[len] != 0
        len = len + 1;
    wend
    len;
end function

function strcmp(s1 as string, s2 as string) as bool
    dim s1len as int
    dim s2len as int
    dim result as bool
    dim count as int
    s1len = strlen(s1);
    s2len = strlen(s2);
    if s1len != s2len then
        result = false;
    else
        count = 0;
        result = true;
        while count != s1len
            if s1[count] != s2[count] then
                result = false;
            end if
            count = count + 1;
        wend
    end if
    result;
end function

function find_object_member_by_name(object as string, name as string) as string
    dim result as string
    dim count as int
    dim tmp_result as bool

    result = 0;
    count = 0;
    while object[count] != 0
        tmp_result = strcmp(object[count][0], name);
        if tmp_result == true then
            result = object[count] + 16;
        end if
        count = count + 1;
    wend
    result;
end function

function init_object_member(name as string, value as int, is_public as bool)
    dim member as string(,,,)
    member = "   ";
    member[0] = name;
    member[1] = value;
    member[2] = is_public;
    member;
end function

function read_op()
    dim op as char
    op = stdin();
    stdin();
    stdin();
    op;
end function

class Calculator
    public dim first as int
    public dim second as int
    public dim op as char

    public function calculate() as res
        dim a as int
        dim b as int
        dim op as char
        dim res as int
        a = this.first;
        b = this.second;
        op = this.op;
        if op == '+' then
            res = sum(a, b);
        end if
        if op == '-' then
              res = sub(a, b);
        end if
        if op == '*' then
            res = mul(a, b);
        end if
        if op == '/' then
            res = div(a, b);
        end if
        res;
    end function
end class

function main()
    dim calc as string

    calc = Calculator();

    while true
        calc.first = read_num();
        calc.op = read_op();
        calc.second = read_num();
        print_num(calc.calculate());
    wend
end function
