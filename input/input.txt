function read_op()
    dim op as char
    op = stdin();
    stdin();
    stdin();
    op;
end function

class Calculator
    public dim first as int
    public dim second as int
    public dim op as char

    public function calculate() as res
        dim a as int
        dim b as int
        dim op as char
        dim res as int
        a = this.first;
        b = this.second;
        op = this.op;
        if op == '+' then
            res = sum(a, b);
        end if
        if op == '-' then
              res = sub(a, b);
        end if
        if op == '*' then
            res = mul(a, b);
        end if
        if op == '/' then
            res = div(a, b);
        end if
        res;
    end function
end class

function main()
    dim calc as string

    calc = Calculator();

    while true
        calc.first = read_num();
        calc.op = read_op();
        calc.second = read_num();
        print_num(calc.calculate());
    wend
end function
